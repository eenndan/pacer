litgen_find_nanobind()

# THE CODEGEN RUNS WHEN ITS INPUTS CHANGE, NOT ON EVERY BUILD (HEALTH-1, 2026-09-25). It was an
# `add_custom_target`, which CMake treats as always out of date: every `pixi run build` — and so
# every `pixi run studio`, which depends on it — re-ran litgen + black on unchanged headers
# (1.5 s of a no-op build, 3.5 s every other one, when the rewritten nanobind_pacer.cpp then
# recompiled). Its inputs: the headers the generator reads (all of pacer/, so a header it starts
# reading needs no edit here; CONFIGURE_DEPENDS, because `pixi run build` no longer re-configures
# an existing build dir, and a new header must still join the list), the generator with its litgen
# options, and pixi.lock, which pins the litgen and black that shape the output. The generator
# writes a file only when its bytes change.
#
# The generated files are NOT declared as this command's outputs: they are committed sources, and a
# `clean` deletes every output. The stamp stands for them instead, and OBJECT_DEPENDS recompiles
# nanobind_pacer.cpp after every codegen run, in the same build.
#
# CI regenerates UNCONDITIONALLY before its drift gate (`pixi run gen-bindings`, ci.yml), so the
# gate never rests on this dependency tracking.
file(GLOB_RECURSE _pacer_binding_headers CONFIGURE_DEPENDS "${CMAKE_SOURCE_DIR}/pacer/*.hpp")
set(_pacer_binding_inputs ${_pacer_binding_headers} "${CMAKE_CURRENT_SOURCE_DIR}/generate-bindings.py")
if(EXISTS "${CMAKE_SOURCE_DIR}/pixi.lock")
    list(APPEND _pacer_binding_inputs "${CMAKE_SOURCE_DIR}/pixi.lock")
endif()
set(_pacer_bindings_stamp "${CMAKE_CURRENT_BINARY_DIR}/pacer_bindings.stamp")
add_custom_command(
    OUTPUT "${_pacer_bindings_stamp}"
    COMMAND "${Python_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/generate-bindings.py"
    COMMAND "${CMAKE_COMMAND}" -E touch "${_pacer_bindings_stamp}"
    DEPENDS ${_pacer_binding_inputs}
    COMMENT "Building pacer python bindings"
    VERBATIM
)
add_custom_target(_generate_pacer_bindings DEPENDS "${_pacer_bindings_stamp}")

# The python module sources
set(python_module_sources
    module.cpp                # The python module entry point
    nanobind_pacer.cpp        # The pybind11 bindings to the library, which are mainly auto-generated by litgen
)
set_source_files_properties(nanobind_pacer.cpp PROPERTIES OBJECT_DEPENDS "${_pacer_bindings_stamp}")

nanobind_add_module(_pacer ${python_module_sources})

add_library(pacer INTERFACE)
target_link_libraries(pacer INTERFACE
  pacer::datatypes
  pacer::geometry
  pacer::laps
  pacer::gps-source
)

if (SKBUILD)
  set(LITGEN_PATH_PACER "${CMAKE_INSTALL_PREFIX}" CACHE PATH "Path to install the pacer python module")
else()
  set(LITGEN_PATH_PACER  "${CMAKE_SOURCE_DIR}/bindings/pacer" CACHE PATH "Path to install the pacer python module")
endif()

# Call litgen_setup_module to generate the python wrapper around the native module
litgen_setup_module(
    # The C++ library for which we are building bindings
    pacer
    # The native python module name
    _pacer
    # This is the python wrapper around the native module
    pacer
    # The path where the python module will be copied after build (for editable mode)
    "${LITGEN_PATH_PACER}"
)

add_dependencies(
    _pacer
    _generate_pacer_bindings
)
